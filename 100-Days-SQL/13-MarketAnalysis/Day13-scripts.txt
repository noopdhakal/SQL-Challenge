-- Create USERS table
CREATE TABLE users (
    user_id         NUMBER,
    join_date       DATE,
    favorite_brand  VARCHAR2(50)
);

-- Create ORDERS table
CREATE TABLE orders (
    order_id        NUMBER,
    order_date      DATE,
    item_id         NUMBER,
    buyer_id        NUMBER,
    seller_id       NUMBER
);

-- Create ITEMS table
CREATE TABLE items (
    item_id         NUMBER,
    item_brand      VARCHAR2(50)
);
